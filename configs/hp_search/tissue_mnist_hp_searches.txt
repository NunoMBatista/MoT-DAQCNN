# TissueMNIST Hyperparameter Search — All Architectures & Kernel Variants
# =========================================================================

# This file contains commands to run hyperparameter searches for all
# architectures on the TissueMNIST dataset, including single-kernel baselines

#   python3 experiments/batch_runner.py configs/hp_search/tissue_mnist_hp_searches.txt

#   python3 experiments/batch_runner.py --dry-run configs/hp_search/tissue_mnist_hp_searches.txt
#

# =============================================================================
# Original DAQCNN — Single Kernels (Baselines)

python experiments/hyperparameter_search.py --config configs/tissue_mnist_single_kernel_kings_1_kernel_3x3.yml --search-config configs/hp_search/tissue_mnist_single_kernel.yml --n-trials 200

# Single kernel: Horizontal only
python experiments/hyperparameter_search.py --config configs/tissue_mnist_single_kernel_horizontal_1_kernel_3x3.yml --search-config configs/hp_search/tissue_mnist_single_kernel.yml --n-trials 200
